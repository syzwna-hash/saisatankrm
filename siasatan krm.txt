    <script type="module">
        import { initializeApp } from "https://www.gstatic.com/firebasejs/11.6.1/firebase-app.js";
        import { getAuth, signInAnonymously, signInWithCustomToken, onAuthStateChanged } from "https://www.gstatic.com/firebasejs/11.6.1/firebase-auth.js";
        import { getFirestore, doc, setDoc, getDocs, collection, query, deleteDoc, getDoc } from "https://www.gstatic.com/firebasejs/11.6.1/firebase-firestore.js";

        // Global configuration from environment
        const appId = typeof __app_id !== 'undefined' ? __app_id : 'haccpkrm-field-pro';
        const firebaseConfig = JSON.parse(__firebase_config);
        const app = initializeApp(firebaseConfig);
        const auth = getAuth(app);
        const db = getFirestore(app);

        let currentUser = null;

        // Rule 3: Auth Before Queries
        async function initAuth() {
            try {
                if (typeof __initial_auth_token !== 'undefined' && __initial_auth_token) {
                    await signInWithCustomToken(auth, __initial_auth_token);
                } else {
                    await signInAnonymously(auth);
                }
            } catch (err) { console.error("Ralat pengesahan:", err); }
        }

        onAuthStateChanged(auth, (user) => {
            currentUser = user;
            if (user) console.log("HACCP Cloud Connected:", user.uid);
        });

        initAuth();

        // Database Functions
        window.saveToCloud = async function() {
            if (!currentUser) { showToast("Menyambung ke pelayan...", "error"); return; }
            const data = getFullData();
            const docId = data.basic.namaPremis ? data.basic.namaPremis.replace(/\s+/g, '-').toLowerCase() + '-' + Date.now() : 'draft-' + Date.now();
            try {
                // Rule 1: Strict Paths
                const docRef = doc(db, 'artifacts', appId, 'users', currentUser.uid, 'investigations', docId);
                await setDoc(docRef, data);
                showToast("Siasatan berjaya disimpan ke awan!", "success");
            } catch (error) { showToast("Gagal menyimpan draf.", "error"); }
        };

        window.loadDrafts = async function() {
            if (!currentUser) return;
            const container = document.getElementById('draftsList');
            container.innerHTML = '<p class="text-center p-4 text-xs text-emerald-600 font-bold">Memuatkan data dari awan...</p>';
            try {
                const colRef = collection(db, 'artifacts', appId, 'users', currentUser.uid, 'investigations');
                const querySnapshot = await getDocs(colRef);
                container.innerHTML = '';
                if (querySnapshot.empty) { container.innerHTML = '<p class="text-center p-4 text-gray-500 text-xs">Tiada draf dijumpai.</p>'; return; }
                const docs = [];
                querySnapshot.forEach(d => docs.push({id: d.id, ...d.data()}));
                // Rule 2: Sort in memory
                docs.sort((a, b) => new Date(b.updatedAt) - new Date(a.updatedAt));
                docs.forEach((data) => {
                    const div = document.createElement('div');
                    div.className = "p-3 border-b hover:bg-gray-50 flex justify-between items-center";
                    div.innerHTML = `
                        <div class="flex-1 mr-2">
                            <p class="font-bold text-sm text-emerald-800 truncate">${data.basic.namaPremis || 'Tanpa Nama'}</p>
                            <p class="text-[9px] text-gray-400">${new Date(data.updatedAt).toLocaleString()}</p>
                        </div>
                        <div class="flex gap-1">
                            <button type="button" onclick="applyDraft('${data.id}')" class="bg-emerald-600 text-white px-2 py-1 rounded text-[10px] font-bold uppercase shadow-sm">Buka</button>
                            <button type="button" onclick="deleteDraft('${data.id}')" class="bg-red-50 text-red-600 px-2 py-1 rounded text-[10px] font-bold uppercase border border-red-200">Padam</button>
                        </div>
                    `;
                    container.appendChild(div);
                });
            } catch (error) { showToast("Ralat memuatkan senarai.", "error"); }
        };

        window.applyDraft = async function(id) {
            try {
                const docRef = doc(db, 'artifacts', appId, 'users', currentUser.uid, 'investigations', id);
                const snap = await getDoc(docRef);
                if (snap.exists()) {
                    const data = snap.data();
                    // Clear current dynamic containers
                    document.getElementById('handlersContainer').innerHTML = '';
                    document.getElementById('rawMaterialsContainer').innerHTML = '';
                    document.getElementById('processContainer').innerHTML = '';
                    
                    const form = document.getElementById('haccpForm');
                    // Populate basic form fields
                    for (const key in data.basic) {
                        const input = form.elements[key];
                        if (input) {
                            if (input.type === 'checkbox') input.checked = data.basic[key];
                            else input.value = data.basic[key];
                        }
                    }
                    // Restore dynamic rows
                    if(data.handlers) data.handlers.forEach(h => addHandlerRow(h.nama, h.jawatan, h.typhoidDate));
                    if(data.rawMaterials) data.rawMaterials.forEach(r => addRawMaterialRow(r.name, r.q1, r.q2, r.q3, r.sumber, r.simpanan, r.keadaan, r.tarikhBeli));
                    if(data.processes) data.processes.forEach(p => addProcessRow(p.step, p.findings, p.q1, p.q2, p.q3, p.q4, p.ulasan));
                    
                    window.generalImages = data.images || [];
                    renderImagePreview();
                    closeModal('loadModal');
                    showToast("Draf berjaya dimuatkan!", "success");
                    updateAgenOptions();
                }
            } catch (err) { showToast("Gagal muat draf.", "error"); }
        };

        window.deleteDraft = async function(id) {
            if(confirm("Padam draf ini secara kekal?")) {
                try {
                    await deleteDoc(doc(db, 'artifacts', appId, 'users', currentUser.uid, 'investigations', id));
                    loadDrafts();
                } catch (err) { showToast("Ralat memadam draf.", "error"); }
            }
        };

    <script>
        const bacteriaData = { "1-6": ["S. aureus", "B. cereus (Emetic)", "B. subtilis"], "7-12": ["C. perfringens", "B. cereus (Diarrheal)", "S. faecalis"], "13-72": ["Salmonella species", "Vibrio parahaemolyticus", "Escherichia coli", "Shigella", "Campylobacter jejuni"], ">4": ["Listeria monocytogenes", "Clostridium botulinum"] };
        const riskTitles = ["Pengendali Makanan", "Pencemaran Silang", "Kemudahan Mencuci Tangan", "Kawalan Proses", "Program Pencucian", "Faktor Luar Kawalan", "Kawalan Makhluk Perosak", "Penyimpanan", "Penyelenggaraan Peralatan/Bangunan", "Kawalan Bahan Kimia", "Kenderaan Pengangkutan"];
        
        let currentStep = 1;
        const totalSteps = 10;
        window.generalImages = [];

        function init() {
            renderDots();
            renderRiskFactors();
            updateAgenOptions();
            if(document.getElementById('handlersContainer').children.length === 0) addHandlerRow();
            if(document.getElementById('rawMaterialsContainer').children.length === 0) addRawMaterialRow();
            if(document.getElementById('processContainer').children.length === 0) addProcessRow();
        }

        window.init = init;

        function renderDots() {
            const dots = document.getElementById('dotsContainer');
            dots.innerHTML = '';
            for(let i=1; i<=totalSteps; i++) {
                const d = document.createElement('div');
                d.id = `dot-${i}`;
                d.className = `h-1 rounded-full transition-all ${i===1 ? 'w-4 bg-emerald-600' : 'w-1 bg-gray-200'}`;
                dots.appendChild(d);
            }
        }

        function renderRiskFactors() {
            const container = document.getElementById('dynamicRiskFactors');
            container.innerHTML = '';
            riskTitles.forEach((title, idx) => {
                const div = document.createElement('div');
                div.className = "risk-section";
                div.innerHTML = `<div class="flex justify-between items-center mb-1"><h3 class="text-xs font-black uppercase text-emerald-800">${idx+1}. ${title}</h3><input type="checkbox" name="chk_risk_${idx+1}" class="w-4 h-4"></div><textarea name="ulasan_risk_${idx+1}" rows="2" class="form-input text-xs" placeholder="Penemuan..."></textarea>`;
                container.appendChild(div);
            });
        }

        function updateAgenOptions() {
            const inkVal = document.getElementById('inkubasiSelect').value;
            const select = document.getElementById('agenSelect');
            if(!select) return;
            select.innerHTML = '<option value="">-- Pilih Bakteria --</option>';
            (bacteriaData[inkVal] || []).forEach(a => {
                const opt = document.createElement('option');
                opt.value = a; opt.innerText = a; select.appendChild(opt);
            });
        }

        function changeStep(n) {
            const newStep = currentStep + n;
            if (newStep < 1 || newStep > totalSteps) return;
            document.getElementById(`step${currentStep}`).classList.remove('active');
            currentStep = newStep;
            document.getElementById(`step${currentStep}`).classList.add('active');
            document.getElementById('stepIndicator').innerText = `Langkah ${currentStep}/${totalSteps}`;
            document.getElementById('prevBtn').disabled = currentStep === 1;
            document.getElementById('nextBtn').innerText = currentStep === totalSteps ? 'SELESAI' : 'LANJUT';
            for(let i=1; i<=totalSteps; i++) {
                const d = document.getElementById(`dot-${i}`);
                if(d) d.className = i <= currentStep ? (i === currentStep ? 'h-1 w-4 bg-emerald-600 rounded-full' : 'h-1 w-2 bg-emerald-400 rounded-full') : 'h-1 w-1 bg-gray-200 rounded-full';
            }
            if(currentStep === totalSteps) generateTemplateKesimpulan();
            window.scrollTo(0, 0);
        }

        window.getFullData = () => {
            const form = document.getElementById('haccpForm');
            const basic = {};
            new FormData(form).forEach((v, k) => basic[k] = v);
            const handlers = [];
            document.querySelectorAll('.handler-row').forEach(row => {
                const i = row.querySelectorAll('input, select');
                handlers.push({ nama: i[0].value, jawatan: i[1].value, typhoidDate: i[2].value });
            });
            const rawMaterials = [];
            document.querySelectorAll('.raw-row').forEach(row => {
                const n = row.querySelector('input').value;
                const s = row.querySelectorAll('select');
                const inp = row.querySelectorAll('input');
                rawMaterials.push({ name: n, q1: s[0].value, q2: s[1].value, q3: s[2].value, sumber: inp[1].value, tarikhBeli: inp[2].value, simpanan: inp[3].value });
            });
            const processes = [];
            document.querySelectorAll('.process-row').forEach(row => {
                const s = row.querySelector('input').value;
                const f = row.querySelector('textarea').value;
                const sel = row.querySelectorAll('select');
                processes.push({ step: s, findings: f, q1: sel[0].value, q2: sel[1].value, q3: sel[2].value, q4: sel[3].value });
            });
            return { basic, handlers, rawMaterials, processes, images: window.generalImages, updatedAt: new Date().toISOString() };
        };

        }

        function previewGeneralImages() {
            const input = document.getElementById('generalImageInput');
            for (const file of input.files) {
                const r = new FileReader();
                r.onload = (e) => { window.generalImages.push({ url: e.target.result }); renderImagePreview(); };
                r.readAsDataURL(file);
            }
        }

        function renderImagePreview() {
            const p = document.getElementById('generalImagePreview');
            p.innerHTML = '';
            window.generalImages.forEach((img, i) => {
                const div = document.createElement('div');
                div.className = "card-dynamic p-1 relative";
                div.innerHTML = `<img src="${img.url}" class="w-full aspect-square object-cover rounded-lg"><button type="button" onclick="window.generalImages.splice(${i},1);renderImagePreview();" class="absolute top-1 right-1 bg-red-600 text-white rounded-full w-5 h-5 flex items-center justify-center font-bold shadow-md">&times;</button>`;
                p.appendChild(div);
            });
        }

        async function generatePDF() {
            const { jsPDF } = window.jspdf;
            const doc = new jsPDF();
            const data = getFullData();
            const d = data.basic;
            doc.setFontSize(14); doc.setFont("helvetica", "bold");
            doc.text("LAPORAN SIASATAN KERACUNAN MAKANAN (HACCPKRM)", 105, 18, { align: "center" });
            doc.line(15, 22, 195, 22);
            doc.autoTable({ startY: 28, body: [["Premis", d.namaPremis || '-'], ["Negeri/Daerah", `${d.negeri} / ${d.daerah}`], ["Siasatan", d.tarikhSiasatan || '-'], ["Kebersihan Sblm", d.kebersihanSebelum || '-'], ["Kebersihan Semasa", d.kebersihanSemasa || '-']], styles: { fontSize: 9 } });
            doc.save(`HACCPKRM_${d.namaPremis || 'Laporan'}.pdf`);
        }
